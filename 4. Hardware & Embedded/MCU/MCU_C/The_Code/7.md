{

#include <stdio.h>
#include "NUC100Series.h"
#include "GPIO.h"
#include "SYS.h"
#include "Seven_Segment.h"
#include "Scankey.h"

#define BUZZER PB11

void Buzzer_Beep(int times)
{
    int i;

    for (i = 0; i < times; i++)
    {
        BUZZER = 0;
        CLK_SysTickDelay(100000);

        BUZZER = 1;
        CLK_SysTickDelay(100000);
    }
}

void ShowBinary(int number)
{
    int i;
    int bit;

    for (i = 3; i >= 0; i--)
    {
        bit = (number >> i) & 1;

        ShowSevenSegment(i, bit);

        CLK_SysTickDelay(2000);

        CloseSevenSegment();
    }
}

int main(void)
{
    int key;
    int number = 0;

    SYS_Init();

    OpenKeyPad();
    OpenSevenSegment();

    GPIO_SetMode(PB, BIT11, GPIO_PMD_OUTPUT);
    BUZZER = 1;

    while (1)
    {
        key = ScanKey();

        if (key >= 1 && key <= 6)
        {
            number = key;

            Buzzer_Beep(key);
        }

        if (number >= 1 && number <= 6)
        {
            ShowBinary(number);
        }
    }
}

{

void Buzz(int times)
{ 
    for (int i = 0; i < times; i++)
    {
        BUZZER = 0;
        CLK_SysTickDelay(2000);
        
        BUZZER = 1;
        CLK_SysTickDelay(2000);
    }
}

int main(void)
{
    int key = 0;
    int last_key = 0;

    Sys_Init();

    OpenSevenSegment();
    OpenKeyPad();

    GPIO_SetMode(PB, BIT11, GPIO_MODE_OUTPUT);      
    
    BUZZER = 1;

    while(1)
    {
        key = Scankey();

        if (last_key >= 1 && last_key <= 6 && key == 0)
        {
            ShowBinary(last_key);
            Buzz(last_key);
        }

        last_key = key;
    }
