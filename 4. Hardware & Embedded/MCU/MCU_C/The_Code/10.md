#include <stdio.h>

#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"

volatile int led_position = 0;
volatile int direction = 0;
volatile uint32_t led_timer = 0;

volatile int red = 0;
volatile int green = 0;
volatile int blue = 0;

volatile int pressed_key = 0;


/* Update RGB LED */
void RGB_Update(void)
{
    if (red)
        PA12 = 0;
    else
        PA12 = 1;

    if (green)
        PA13 = 0;
    else
        PA13 = 1;

    if (blue)
        PA14 = 0;
    else
        PA14 = 1;
}


/* Initialize RGB LED */
void Init_RGB(void)
{
    GPIO_SetMode(PA, BIT12, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT13, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT14, GPIO_MODE_OUTPUT);

    // RGB OFF
    PA12 = 1;
    PA13 = 1;
    PA14 = 1;
}


/* Initialize running LEDs */
void Init_LED(void)
{
    GPIO_SetMode(PC, BIT12, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PC, BIT13, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PC, BIT14, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PC, BIT15, GPIO_MODE_OUTPUT);

    // LEDs OFF
    PC12 = 1;
    PC13 = 1;
    PC14 = 1;
    PC15 = 1;
}


/* Show current running LED */
void Show_Running_LED(void)
{
    // Turn all LEDs OFF
    PC12 = 1;
    PC13 = 1;
    PC14 = 1;
    PC15 = 1;

    // Turn current LED ON
    if (led_position == 0)
        PC12 = 0;
    else if (led_position == 1)
        PC13 = 0;
    else if (led_position == 2)
        PC14 = 0;
    else if (led_position == 3)
        PC15 = 0;
}


/* Move LED one position */
void Move_Running_LED(void)
{
    if (direction == 0)
    {
        // RIGHT
        if (led_position == 3)
            led_position = 0;
        else
            led_position++;
    }
    else
    {
        // LEFT
        if (led_position == 0)
            led_position = 3;
        else
            led_position--;
    }

    Show_Running_LED();
}


/* SysTick interrupt: every 1 ms */
void SysTick_Handler(void)
{
    led_timer++;

    // 1000 ms = 1 second
    if (led_timer >= 1000)
    {
        led_timer = 0;
        Move_Running_LED();
    }
}


/* Initialize keypad */
void Init_Keypad(void)
{
    // Reading lines
    GPIO_SetMode(PA, BIT0, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT1, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT2, GPIO_MODE_QUASI);

    // Scanning lines
    GPIO_SetMode(PA, BIT3, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT4, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT5, GPIO_MODE_QUASI);

    // Idle state
    PA3 = 0;
    PA4 = 0;
    PA5 = 0;

    PA0 = 1;
    PA1 = 1;
    PA2 = 1;

    // Enable interrupt on PA0~PA2
    GPIO_EnableInt(PA, 0, GPIO_INT_BOTH_EDGE);
    GPIO_EnableInt(PA, 1, GPIO_INT_BOTH_EDGE);
    GPIO_EnableInt(PA, 2, GPIO_INT_BOTH_EDGE);

    NVIC_EnableIRQ(GPAB_IRQn);

    // Debounce
    GPIO_SET_DEBOUNCE_TIME(
        GPIO_DBCLKSRC_LIRC,
        GPIO_DBCLKSEL_64
    );

    GPIO_ENABLE_DEBOUNCE(
        PA,
        BIT0 | BIT1 | BIT2
    );
}


/* Read keypad */
int GetKey(void)
{
    int key = 0;

    // Row 1: 1 2 3
    PA3 = 0;
    PA4 = 1;
    PA5 = 1;

    if (PA0 == 0)
        key = 1;
    else if (PA1 == 0)
        key = 2;
    else if (PA2 == 0)
        key = 3;

    // Row 2: 4 5 6
    if (key == 0)
    {
        PA3 = 1;
        PA4 = 0;
        PA5 = 1;

        if (PA0 == 0)
            key = 4;
        else if (PA1 == 0)
            key = 5;
        else if (PA2 == 0)
            key = 6;
    }

    // Row 3: 7 8 9
    if (key == 0)
    {
        PA3 = 1;
        PA4 = 1;
        PA5 = 0;

        if (PA0 == 0)
            key = 7;
        else if (PA1 == 0)
            key = 8;
        else if (PA2 == 0)
            key = 9;
    }

    // Return to idle state
    PA3 = 0;
    PA4 = 0;
    PA5 = 0;

    PA0 = 1;
    PA1 = 1;
    PA2 = 1;

    return key;
}


/* Keypad interrupt */
void GPAB_IRQHandler(void)
{
    int key;

    // Clear interrupt flags
    if (PA->ISRC & BIT0)
        PA->ISRC = BIT0;

    if (PA->ISRC & BIT1)
        PA->ISRC = BIT1;

    if (PA->ISRC & BIT2)
        PA->ISRC = BIT2;

    key = GetKey();

    // Key pressed
    if (key != 0)
    {
        pressed_key = key;
    }
    else
    {
        // Key released
        if (pressed_key == 1)
            red = !red;
        else if (pressed_key == 2)
            green = !green;
        else if (pressed_key == 3)
            blue = !blue;

        RGB_Update();

        pressed_key = 0;
    }
}


/* INT1: change LED direction */
void EINT1_IRQHandler(void)
{
    GPIO_CLR_INT_FLAG(PB, BIT15);

    if (direction == 0)
        direction = 1;
    else
        direction = 0;
}


/* Initialize INT1 */
void Init_INT1(void)
{
    GPIO_SetMode(
        PB,
        BIT15,
        GPIO_MODE_INPUT
    );

    GPIO_EnableEINT1(
        PB,
        15,
        GPIO_INT_RISING
    );

    NVIC_EnableIRQ(EINT1_IRQn);

    // Debounce
    GPIO_SET_DEBOUNCE_TIME(
        GPIO_DBCLKSRC_LIRC,
        GPIO_DBCLKSEL_64
    );

    GPIO_ENABLE_DEBOUNCE(
        PB,
        BIT15
    );
}


int main(void)
{
    SYS_Init();

    Init_LED();
    Init_RGB();
    Init_Keypad();
    Init_INT1();

    Show_Running_LED();

    // SysTick: 1 interrupt every 1 ms
    SysTick_Config(SystemCoreClock / 1000);

    // Enable global interrupts
    __enable_irq();

    while (1)
    {
        // Everything is handled by interrupts
    }
}
