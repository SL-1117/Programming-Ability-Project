#include <stdio.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "Seven_Segment.h"
#include "Scankey.h"

#define B 7
#define C 8
#define PLUS 9

int num1 = 0;
int num2 = 0;
int state = 0;

void Refresh_7seg(int value)
{
    static int digit = 0;

    int d0 = value % 10;
    int d1 = (value / 10) % 10;
    int d2 = (value / 100) % 10;
    int d3 = (value / 1000) % 10;

    CloseSevenSegment();

    if (digit == 0 && value != 0)
        ShowSevenSegment(0, d0);

    else if (digit == 1 && value >= 10)
        ShowSevenSegment(1, d1);

    else if (digit == 2 && value >= 100)
        ShowSevenSegment(2, d2);

    else if (digit == 3 && value >= 1000)
        ShowSevenSegment(3, d3);

    CLK_SysTickDelay(1000);

    digit++;

    if (digit >= 4)
        digit = 0;
}

int GetDisplayNumber(void)
{
    if (state == 0)
        return num1;

    if (state == 1)
        return num2;

    return num1 + num2;
}

void AddDigit(int *num, int digit)
{
    *num = (*num * 10 + digit) % 1000;
}

void Backspace(int *num)
{
    *num = *num / 10;
}

void ClearAll(void)
{
    num1 = 0;
    num2 = 0;
    state = 0;
    CloseSevenSegment();
}

int main(void)
{
    int key;
    int lastKey = 0;

    SYS_Init();

    OpenSevenSegment();
    OpenKeyPad();

    while (1)
    {
        key = Scankey();

        if (lastKey >= 1 && lastKey <= 6 && key == 0)
        {
            if (state == 0)
                AddDigit(&num1, lastKey);

            else if (state == 1)
                AddDigit(&num2, lastKey);
        }

        if (lastKey == B && key == 0)
        {
            if (state == 0)
                Backspace(&num1);

            else if (state == 1)
                Backspace(&num2);
        }

        if (lastKey == C && key == 0)
        {
            ClearAll();
        }

        if (lastKey == PLUS && key == 0)
        {
            if (state == 0)
                state = 1;

            else if (state == 1)
                state = 2;
        }

        Refresh_7seg(GetDisplayNumber());

        lastKey = key;
    }
}

int main(void)
{
    int key = 0;
    int last_key = 0;

    SYS_Init();

    OpenSevenSegment();
    OpenKeyPad();

    while(1)
    {
        key = Scankey();

        if (last_key >= 1 && last_key <= 6 && key == 0)
        {
            if (state == 0) AddNum(last_key, &num1);
            else if (state == 1) AddNum(last_key, &num2);
        }

        else if (last_key == B && key == 0)
        {
            if (state == 0) BackSpace(&num1);
            else if (state == 1) BackSpace(&num2);
        }

        else if (last_key == C && key == 0)
        {
            ClearAll();
        }

        else if (last_key == PLUS_KEY && key == 0)
        {
            if (state == 0) state = 1;
            else if (state == 1) state = 2;
        }

        Refresh_7seg(GetDisplayNum());

        if (state == 2)
        {
            while (key != C)
            {
                key = Scankey();
            }

            ClearAll();
        }

        last_key = key;
    }
}

    static int digit = 0;

    int d0 = value % 10;
    int d1 = (value / 10) % 10;
    int d2 = (value / 100) % 10;
    int d3 = (value / 1000) % 10;

    CloseSevenSegment();

    if (digit == 0 && value != 0)           ShowSevenSegment(0, d0);
    else if (digit == 1 && value >= 10)     ShowSevenSegment(1, d1);
    else if (digit == 2 && value >= 100)    ShowSevenSegment(2, d2);
    else if (digit == 3 && value >= 1000)   ShowSevenSegment(3, d3);

    CLK_SysTickDelay(1000);

    digit++;
    
    if (digit >= 4) digit = 0;
}
