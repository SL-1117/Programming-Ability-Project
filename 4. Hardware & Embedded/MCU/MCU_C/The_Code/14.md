#include <stdio.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "LCD.h"
#include "Scankey.h"

typedef enum { WAIT_A, WAIT_B, SHOW_RESULT } State;

void show_result(int a, int b)
{
    char buf[17];
    clear_LCD();
    sprintf(buf, "%d + %d = %d", a, b, a + b);  print_Line(0, buf);
    sprintf(buf, "%d - %d = %d", a, b, a - b);  print_Line(1, buf);  
    sprintf(buf, "%d x %d = %d", a, b, a * b);  print_Line(2, buf);
    sprintf(buf, "%d / %d = %d", a, b, a / b);  print_Line(3, buf); 
}

int main(void)
{
    State   state = WAIT_A;
    uint8_t key, pressed_key = 0;
    int     a = 0, b = 0;

    SYS_Init();       
    init_LCD();
    clear_LCD();
    OpenKeyPad();

    while (1) {
        key = ScanKey();              

        if (pressed_key != 0 && key == 0) {
            switch (state) {
            case WAIT_A:
                a = pressed_key;
                clear_LCD();
                printC(0, 0, '0' + a);  
                state = WAIT_B;
                break;

            case WAIT_B:
                b = pressed_key;
                show_result(a, b);
                state = SHOW_RESULT;
                break;

            case SHOW_RESULT:           
                clear_LCD();
                state = WAIT_A;
                break;
            }
        }
        pressed_key = key;
        CLK_SysTickDelay(20000);      
    }
}
