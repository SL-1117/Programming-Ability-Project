#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "GPIO.h"
#include "ADC.h"
#include "LCD.h"
#include "Seven_Segment.h"

uint8_t speed = 1;
uint8_t running = 0;
int8_t dirX = 1;

// Modified
int8_t mov = 1;
#define L_BAR_X 0
#define R_BAR_X (LCD_Xmax - 1)

int16_t x = 60;
int16_t y = 28;

void Init_RGB(void)
{
    GPIO_SetMode(PA, BIT12 | BIT13 | BIT14, GPIO_MODE_OUTPUT);

    PA12 = 1;
    PA13 = 1;
    PA14 = 1;
}

void Update_RGB(void)
{
    PA12 = 1;
    PA13 = 1;
    PA14 = 1;

    if (speed == 1)
        PA12 = 0;
    else if (speed == 2)
        PA13 = 0;
    else
        PA14 = 0;
}

void Init_Button(void)
{
    GPIO_SetMode(
        PA,
        BIT0 | BIT1 | BIT2 | BIT3 | BIT4 | BIT5,
        GPIO_MODE_QUASI
    );

    PA0 = 1;
    PA1 = 0;
    PA2 = 1;
    PA3 = 1;
    PA4 = 1;
    PA5 = 1;
}

uint8_t Read_Key5(void)
{
    PA0 = 1;
    PA1 = 0;
    PA2 = 1;

    return (PA4 == 0);
}

void Init_ADC(void)
{
    ADC_Open(
        ADC,
        ADC_INPUT_MODE_SINGLE_END,
        ADC_OPERATION_MODE_SINGLE_CYCLE,
        ADC_CH_7_MASK
    );

    ADC_POWER_ON(ADC);
    // ADC_EnableInt(ADC, ADC_ADF_INT);
    // NVIC_EnableIRQ(ADC_IRQn);
}

uint32_t Read_ADC(void)
{
    uint32_t timeout = 100000;
    uint32_t value;

    ADC_CLR_INT_FLAG(ADC, ADC_ADF_INT);
    ADC_START_CONV(ADC);

    while ((ADC_GET_INT_FLAG(ADC, ADC_ADF_INT) & ADC_ADF_INT) == 0)
    {
        if (timeout == 0)
            return 0xFFFFFFFF;

        timeout--;
    }

    value = ADC_GET_CONVERSION_DATA(ADC, 7);

    ADC_CLR_INT_FLAG(ADC, ADC_ADF_INT);

    return value;
}

void Update_Speed(void)
{
    uint32_t value;
    uint8_t new_speed;

    value = Read_ADC();

    if (value == 0xFFFFFFFF)
        return;

    if (value <= 1364)
        new_speed = 1;
    else if (value <= 2729)
        new_speed = 2;
    else
        new_speed = 3;

    if (new_speed != speed)
    {
        speed = new_speed;

        Update_RGB();

        CloseSevenSegment();
        ShowSevenSegment(0, speed);
    }
}

uint32_t Get_Move_Time(void)
{
    if (speed == 1)
        return 240;

    if (speed == 2)
        return 120;

    return 60;
}

void Draw_Square(int16_t x, int16_t y, uint8_t color)
{
    int16_t i;
    int16_t j;

    for (j = 0; j < 8; j++)
    {
        for (i = 0; i < 8; i++)
        {
            draw_Pixel(x + i, y + j, color, 0);
        }
    }
}

void Move_Square(void)
{
    draw(x, y, 0);

    x += mov * dirX;

    if      (x     <= L_BAR_X) { x = L_BAR_X    ; dirX =  1; }
    else if (x + 7 >= R_BAR_X) { x = R_BAR_X - 7; dirX = -1; }

    draw(x, y, 1);
}

int32_t main(void)
{
    uint8_t button_ready = 1;
    uint32_t adc_time = 0;
    uint32_t move_time = 0;

    SYS_Init();

    OpenSevenSegment();

    Init_RGB();
    Init_Button();

    init_LCD();
    clear_LCD();

    Draw_Square(x, y, 1);

    Init_ADC();

    Update_Speed();
    Update_RGB();

    CloseSevenSegment();
    ShowSevenSegment(0, speed);

    while (1)
    {
        CLK_SysTickDelay(10000);

        // Actually this is js the debounce system, don't be afraid
        if (Read_Key5() == 1 && button_ready == 1)
        {
            CLK_SysTickDelay(20000);

            if (Read_Key5() == 1)
            {
                running = !running;
                button_ready = 0;
                move_time = 0;
            }
        }

        if (Read_Key5() == 0)
        {
            button_ready = 1;
        }

        // 
        adc_time += 10;

        if (adc_time >= 50)
        {
            adc_time = 0;
            Update_Speed();
        }

        // Control the speed of the movement
        if (running == 1)
        {
            move_time += 10;

            if (move_time >= Get_Move_Time())
            {
                move_time = 0;
                Move_Square();
            }
        }
    }
}
