```c
#include <stdio.h>
#include <stdint.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "GPIO.h"
#include "ADC.h"
#include "LCD.h"
#include "Seven_Segment.h"

#define FRAME_COUNT     4
#define SPRITE_W        16
#define SPRITE_H        24

#define SPRITE_X        56
#define SPRITE_Y        20

#define ADC_INTERVAL    50
#define NORMAL_SPEED    200
#define FAST_SPEED      80

const uint16_t WALK[FRAME_COUNT][SPRITE_H] = {
    {
        0x0380, 0x07C0, 0x07C0, 0x07C0, 0x0380, 0x0100,
        0x0100, 0x0380, 0x0540, 0x0920, 0x0920, 0x1110,
        0x0100, 0x0100, 0x0280, 0x0280, 0x0440, 0x0440,
        0x0820, 0x0820, 0x1010, 0x1010, 0x2008, 0x300C
    },
    {
        0x0000, 0x0380, 0x07C0, 0x07C0, 0x07C0, 0x0380,
        0x0100, 0x0100, 0x0380, 0x0540, 0x0540, 0x0920,
        0x0100, 0x0100, 0x0100, 0x0280, 0x0280, 0x0440,
        0x0440, 0x0440, 0x0820, 0x0820, 0x0820, 0x0C30
    },
    {
        0x0380, 0x07C0, 0x07C0, 0x07C0, 0x0380, 0x0100,
        0x0100, 0x0100, 0x0180, 0x0140, 0x0140, 0x0120,
        0x0100, 0x0100, 0x0180, 0x0140, 0x0140, 0x0120,
        0x0120, 0x0140, 0x0140, 0x0100, 0x0100, 0x0180
    },
    {
        0x0380, 0x07C0, 0x07C0, 0x07C0, 0x0380, 0x0100,
        0x0100, 0x0380, 0x0540, 0x0540, 0x0920, 0x0100,
        0x0100, 0x0100, 0x0280, 0x0240, 0x0220, 0x0410,
        0x0410, 0x0420, 0x0820, 0x0820, 0x0800, 0x0C00
    }
};

uint32_t adc_value = 0;

uint8_t light_state = 255;
uint8_t current_frame = 0;

uint32_t adc_time = 0;
uint32_t animation_time = 0;

// 0 = Red, 1 = Green, 2 = Yellow

void Init_RGB(void)
{
    GPIO_SetMode(PA, BIT12 | BIT13 | BIT14, GPIO_MODE_OUTPUT);

    PA12 = 1;
    PA13 = 1;
    PA14 = 1;
}

void Update_RGB(uint8_t state)
{
    PA12 = 1;
    PA13 = 1;
    PA14 = 1;

    if (state == 0)
    {
        // Red
        PA12 = 0;
    }
    else if (state == 1)
    {
        // Green
        PA13 = 0;
    }
    else
    {
        // Yellow: Red + Green
        PA12 = 0;
        PA13 = 0;
    }
}

void Init_ADC(void)
{
    ADC_Open(
        ADC,
        ADC_INPUT_MODE_SINGLE_END,
        ADC_OPERATION_MODE_SINGLE_CYCLE,
        ADC_CH_7_MASK
    );

    ADC_POWER_ON(ADC);
}

uint32_t Read_ADC(void)
{
    uint32_t timeout = 100000;
    uint32_t value;

    ADC_CLR_INT_FLAG(ADC, ADC_ADF_INT);

    ADC_START_CONV(ADC);

    while ((ADC_GET_INT_FLAG(ADC, ADC_ADF_INT) & ADC_ADF_INT) == 0)
    {
        if (timeout == 0)
            return 0xFFFFFFFF;

        timeout--;
    }

    value = ADC_GET_CONVERSION_DATA(ADC, 7);

    ADC_CLR_INT_FLAG(ADC, ADC_ADF_INT);

    return value;
}

void Update_Traffic_Light(void)
{
    uint8_t new_state;

    if (adc_value <= 1364)
        new_state = 0;
    else if (adc_value <= 2729)
        new_state = 1;
    else
        new_state = 2;

    if (new_state != light_state)
    {
        light_state = new_state;

        Update_RGB(light_state);

        animation_time = 0;
    }
}

uint32_t Get_Walk_Interval(void)
{
    if (light_state == 1)
        return NORMAL_SPEED;

    if (light_state == 2)
        return FAST_SPEED;

    return 0;
}

void Draw_Sprite(uint8_t frame)
{
    int16_t x;
    int16_t y;

    uint16_t bitmap;

    for (y = 0; y < SPRITE_H; y++)
    {
        bitmap = WALK[frame][y];

        for (x = 0; x < SPRITE_W; x++)
        {
            if (bitmap & (0x8000 >> x))
                draw_Pixel(SPRITE_X + x, SPRITE_Y + y, 1, 0);
            else
                draw_Pixel(SPRITE_X + x, SPRITE_Y + y, 0, 0);
        }
    }
}

void Update_Animation(void)
{
    uint32_t interval;

    if (light_state == 0)
        return;

    interval = Get_Walk_Interval();

    animation_time += 10;

    if (animation_time >= interval)
    {
        animation_time = 0;

        current_frame++;

        if (current_frame >= FRAME_COUNT)
            current_frame = 0;

        Draw_Sprite(current_frame);
    }
}

void Update_SevenSegment(void)
{
    uint8_t digits[4];
    uint8_t i;
    uint8_t position;
    uint32_t temp;

    temp = adc_value;

    // Separate ADC value into individual digits

    for (i = 0; i < 4; i++)
    {
        digits[i] = temp % 10;
        temp /= 10;
    }

    // Scan four digits

    for (i = 0; i < 4; i++)
    {
        // Position 3 is the rightmost digit,
        // position 0 is the leftmost digit.

        position = 3 - i;

        // Suppress leading zeros

        if (i > 0 && adc_value < (i == 1 ? 10 :
                                  i == 2 ? 100 : 1000))
        {
            CloseSevenSegment();
        }
        else
        {
            ShowSevenSegment(position, digits[i]);
        }

        CLK_SysTickDelay(1000);
    }
}

int32_t main(void)
{
    uint32_t value;

    SYS_Init();

    Init_RGB();
    Init_ADC();

    OpenSevenSegment();

    init_LCD();
    clear_LCD();

    // Read initial VR1 value

    value = Read_ADC();

    if (value != 0xFFFFFFFF)
        adc_value = value;

    // Initialize traffic light

    Update_Traffic_Light();

    // Draw initial pedestrian pose

    Draw_Sprite(current_frame);

    while (1)
    {
        // Refresh seven-segment display

        Update_SevenSegment();

        // Approximately 10 ms per main-loop cycle

        CLK_SysTickDelay(6000);

        // Read ADC every approximately 50 ms

        adc_time += 10;

        if (adc_time >= ADC_INTERVAL)
        {
            adc_time = 0;

            value = Read_ADC();

            if (value != 0xFFFFFFFF && value != adc_value)
            {
                adc_value = value;

                Update_Traffic_Light();
            }
        }

        // Update walking animation

        Update_Animation();
    }
}
