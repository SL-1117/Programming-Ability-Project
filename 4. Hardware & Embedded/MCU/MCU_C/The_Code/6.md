    OpenSevenSegment();

    GPIO_SetMode(PC, BIT14, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT15, GPIO_PMD_OUTPUT);

    {
        key = Scankey();
        
        if (key >= 1 && key <= 7)
        {
            if (key == 1)
            {
                ShowSevenSegment(1, 1);
                PC15 = 1; PC14 = 1; PC13 = 1; PC12 = 0;
            }
            else if (key == 2)
            {
                ShowSevenSegment(1, 3);
                PC15 = 1; PC14 = 1; PC13 = 0; PC12 = 0;
            }
            else if (key == 3)
            {
                ShowSevenSegment(1, 8);
                PC15 = 0; PC14 = 1; PC13 = 1; PC12 = 1;
            }
            else if (key == 4)
            {
                ShowSevenSegment(1, 5);
                PC15 = 1; PC14 = 0; PC13 = 1; PC12 = 0;
            }
            else if (key == 5)
            {
                ShowSevenSegment(1, 3);
                PC15 = 1; PC14 = 1; PC13 = 0; PC12 = 0;

            else if (key == 6)
            {
                ShowSevenSegment(1, 2);
                PC15 = 1; PC14 = 1; PC13 = 0; PC12 = 1;

                ShowSevenSegment(1, 7);

#include <stdio.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include <Seven_Segment.h>
#include <Scankey.h>

int main(void)
{
    int key;
    int number;
    int ID[7] = {1, 3, 8, 5, 3, 2, 7};

    SYS_Init();

    OpenKeyPad();
    OpenSevenSegment();

    GPIO_SetMode(PC, BIT12 | BIT13 | BIT14 | BIT15, GPIO_MODE_OUTPUT);

    while (1)
    {
        key = ScanKey();

        if (key >= 1 && key <= 7)
        {
            number = ID[key - 1];

            ShowSevenSegment(0, number);

            /* LED: LEFT -> RIGHT = 1, 2, 4, 8 */

            if (number & 8)
                PC12 = 0;
            else
                PC12 = 1;

            if (number & 4)
                PC13 = 0;
            else
                PC13 = 1;

            if (number & 2)
                PC14 = 0;
            else
                PC14 = 1;

            if (number & 1)
                PC15 = 0;
            else
                PC15 = 1;
        }
    }
}
