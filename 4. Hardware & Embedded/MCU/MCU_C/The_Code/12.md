#include <stdio.h>
#include <stdlib.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "Seven_Segment.h"
#include "Scankey.h"


/* Segment -> PE bit (from the slide: G E D B A F DOT C = PE7 ... PE0) */
#define S_A  BIT3
#define S_B  BIT4
#define S_C  BIT0
#define S_D  BIT5
#define S_E  BIT6
#define S_F  BIT2
#define S_G  BIT7

/* Digit order: 0 = leftmost, 3 = rightmost (same as Lab5.1).
   If the ring runs in the wrong direction, change to (3 - (i)). */
#define DIG(i)  (i)


volatile uint8_t mode = 0;
// 0 = all on, blinking
// 1 = big ring
// 2 = small ring
// 3 = small square

volatile uint8_t dir = 0;
// 0 = clockwise
// 1 = counter-clockwise

volatile uint8_t step = 0;
volatile uint8_t blink_on = 1;

volatile uint8_t pressed_key = 0;
volatile uint16_t count_250ms = 0;


/* Big ring: 12 positions, clockwise starting from top-left */
const uint8_t big_digit[12] = {0, 1, 2, 3, 3, 3, 3, 2, 1, 0, 0, 0};
const uint8_t big_seg[12]   = {S_A, S_A, S_A, S_A, S_B, S_C,
                               S_D, S_D, S_D, S_D, S_E, S_F};

/* Small ring: 6 positions per digit */
const uint8_t ring_seg[6] = {S_A, S_B, S_C, S_D, S_E, S_F};


void Init_INT1(void)
{
    GPIO_SetMode(PB, BIT15, GPIO_MODE_INPUT);

    GPIO_EnableEINT1(PB, 15, GPIO_INT_FALLING);

    NVIC_EnableIRQ(EINT1_IRQn);

    GPIO_SET_DEBOUNCE_TIME(GPIO_DBCLKSRC_LIRC, GPIO_DBCLKSEL_64);
    GPIO_ENABLE_DEBOUNCE(PB, BIT15);
}


void Init_key(void)
{
    GPIO_SetMode(PA, BIT0, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT1, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT2, GPIO_MODE_QUASI);

    GPIO_SetMode(PA, BIT3, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT4, GPIO_MODE_QUASI);
    GPIO_SetMode(PA, BIT5, GPIO_MODE_QUASI);

    PA0 = 1;
    PA1 = 1;
    PA2 = 1;

    /* PA3 low forever, PA4/PA5 high forever */
    PA3 = 0;
    PA4 = 1;
    PA5 = 1;

    GPIO_EnableInt(PA, 0, GPIO_INT_BOTH_EDGE);
    GPIO_EnableInt(PA, 1, GPIO_INT_BOTH_EDGE);
    GPIO_EnableInt(PA, 2, GPIO_INT_BOTH_EDGE);

    NVIC_EnableIRQ(GPAB_IRQn);

    GPIO_SET_DEBOUNCE_TIME(GPIO_DBCLKSRC_LIRC, GPIO_DBCLKSEL_64);
    GPIO_ENABLE_DEBOUNCE(PA, BIT0 | BIT1 | BIT2);
}


int GetKey(void)
{
    if (PA0 == 0) return 1;
    if (PA1 == 0) return 2;
    if (PA2 == 0) return 3;
    return 0;
}


void GPAB_IRQHandler(void)
{
    int key;

    if (PA->ISRC & BIT0) PA->ISRC = BIT0;
    if (PA->ISRC & BIT1) PA->ISRC = BIT1;
    if (PA->ISRC & BIT2) PA->ISRC = BIT2;

    key = GetKey();

    /* Key pressed: remember it */
    if (key != 0)
    {
        pressed_key = key;
    }

    /* Key released: switch mode */
    else if (pressed_key != 0)
    {
        mode = pressed_key;     // 1 = big, 2 = small, 3 = square
        dir = 0;                // every mode switch -> clockwise
        step = 0;
        count_250ms = 0;

        pressed_key = 0;
    }
}


void EINT1_IRQHandler(void)
{
    GPIO_CLR_INT_FLAG(PB, BIT15);

    /* Reverse direction from the current position */
    dir = !dir;
}


void SysTick_Handler(void)
{
    uint8_t len;

    count_250ms++;

    if (count_250ms >= 250)
    {
        count_250ms = 0;

        if (mode == 0)
        {
            blink_on = !blink_on;
        }
        else
        {
            if (mode == 1)      len = 12;
            else if (mode == 2) len = 6;
            else                len = 8;

            if (dir == 0)
                step = (step + 1) % len;
            else
                step = (step + len - 1) % len;
        }
    }
}


int32_t main(void)
{
    uint8_t seg[4];
    uint8_t m, s, b;
    uint8_t i;

    SYS_Init();

    Init_key();
    Init_INT1();

    OpenSevenSegment();

    SysTick_Config(SystemCoreClock / 1000);   // 1 ms

    while (1)
    {
        /* Take a snapshot so the pattern is consistent */
        m = mode;
        s = step;
        b = blink_on;

        for (i = 0; i < 4; i++)
            seg[i] = 0;

        /* ---------- Build the pattern (1 = segment ON) ---------- */
        if (m == 0)
        {
            /* All on / all off */
            if (b)
            {
                for (i = 0; i < 4; i++)
                    seg[i] = 0xFF;
            }
        }
        else if (m == 1)
        {
            /* Big ring: 1 on, 2 off */
            for (i = 0; i < 12; i++)
            {
                if (((i + 12 - s) % 3) == 0)
                    seg[big_digit[i]] |= big_seg[i];
            }
        }
        else if (m == 2)
        {
            /* Small ring on every digit, synchronized: 1 on, 2 off */
            for (i = 0; i < 6; i++)
            {
                if (((i + 6 - s) % 3) == 0)
                {
                    seg[0] |= ring_seg[i];
                    seg[1] |= ring_seg[i];
                    seg[2] |= ring_seg[i];
                    seg[3] |= ring_seg[i];
                }
            }
        }
        else if (m == 3)
        {
            /* Half digit = small square
               step 0~3 : top half of digit 0~3 (left -> right)
               step 4~7 : bottom half of digit 3~0 (right -> left) */
            if (s < 4)
                seg[s] = S_A | S_B | S_F | S_G;
            else
                seg[7 - s] = S_D | S_C | S_E | S_G;
        }

        /* ---------- Show it (segments are active low) ---------- */
        for (i = 0; i < 4; i++)
        {
            CloseSevenSegment();

            if (seg[i] != 0)
            {
                ShowSevenSegment(DIG(i), 8);   // only used to select the digit
                PE->DOUT = ~seg[i];            // then override the segments
            }
        }
    }
}
