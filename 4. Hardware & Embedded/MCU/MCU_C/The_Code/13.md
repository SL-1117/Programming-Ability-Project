```cpp
#include <stdio.h>
#include <string.h>
#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "LCD.h"
#include "Scankey.h"
#include "Seven_Segment.h"

typedef enum { IDLE, INPUT, SHOWING, DONE } State;

// 字串對應表(大小寫必須完全相同)
const char *str_table[5] = {
    "",                 // 0 不使用
    "hELLo WOrlD",      // 1
    "Latent Defect",    // 2
    "Critical Error",   // 3
    "Ion Efflux"        // 4
};

// 七段:第 i 個輸入要顯示在由左至右第 i 位
// 若實際左右相反,把 3 - i 改成 i
#define SEG_POS(i)  (3 - (i))

uint8_t input_buf[4];   // 存輸入的代號 1~4
uint8_t input_cnt = 0;

// 掃描一輪七段(約 4ms)
void refresh_7seg(void)
{
    uint8_t i;
    for (i = 0; i < 4; i++) {
        CloseSevenSegment();
        if (i < input_cnt)
            ShowSevenSegment(SEG_POS(i), input_buf[i]);
        CLK_SysTickDelay(1000);          // 1ms
    }
    CloseSevenSegment();
}

// 等待 ms 毫秒,期間持續刷新七段
void wait_with_refresh(uint32_t ms)
{
    uint32_t n = ms / 4;
    while (n--) refresh_7seg();
}

// 在第 line 行(0~3)顯示字串:偶數索引(第1、3行)靠左,奇數索引(第2、4行)靠右
void show_line(uint8_t line, const char *s)
{
    int16_t x;
    if (line % 2 == 0)
        x = 0;                          
    else
        x = (16 - strlen(s)) * 8;       
    printS(x, line * 16, (char *)s);
}

void clear_all(void)
{
    input_cnt = 0;
    CloseSevenSegment();
    clear_LCD();
}

int main(void)
{
    State   state = IDLE;
    uint8_t key, pressed_key = 0, k, i;

    SYS_Init();
    init_LCD();
    clear_LCD();
    OpenKeyPad();
    OpenSevenSegment();                
    CloseSevenSegment();

    while (1) {
        key = ScanKey();

        if (pressed_key != 0 && key == 0) {
            k = pressed_key;

            switch (state) {
            case IDLE:
            case DONE:
                if (k == 8) {            
                    clear_all();
                    state = INPUT;
                }
                break;

            case INPUT:
                if (k >= 1 && k <= 4) {
                    if (input_cnt < 4)   // 最多 4 個,超過不接受
                        input_buf[input_cnt++] = k;
                }
                else if (k == 8) {       // 重新開始(視需求可刪除)
                    clear_all();
                }
                else if (k == 9) {
                    if (input_cnt > 0) { // 未輸入就按 9:無動作
                        state = SHOWING;
                        clear_LCD();
                        for (i = 0; i < input_cnt; i++) {
                            show_line(i, str_table[input_buf[i]]);
                            if (i < input_cnt - 1)
                                wait_with_refresh(2000);   // 每隔 2 秒下一行
                        }
                        state = DONE;
                        pressed_key = 0;    // 顯示期間的按鍵全部丟棄
                        continue;
                    }
                }
                break;

            default:
                break;
            }
        }
        pressed_key = key;
        refresh_7seg();                  // 持續刷新七段(同時當作防彈跳延遲)
    }
}
