#include <stdio.h>

#include "MCU.h"
#include "SYS.h"
#include "Scankey.h"
#include "Seven_Segment.h"
#include "GPIO.h"
#include "NCU100Series.h"

int main(void) 
{
    SYS_Init();

    GPIO_SetMode(PC, BIT12, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT13, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT14, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT15, GPIO_PMD_OUTPUT);

    while(1)
    {
        for(int i = 0; i < 6; i++)
        {
            PC15 = 1; PC14 = 1; PC13 = 1; PC12 = 1;

            if (i % 6 == 0) PC15 = 0;
            else if (i % 6 == 1) PC14 = 0;
            else if (i % 6 == 2) PC13 = 0;
            else if (i % 6 == 3) PC12 = 0;
            else if (i % 6 == 4) PC13 = 0;
            else if (i % 6 == 5) PC14 = 0;

            CLK_SysTickDelay(1000000);
        }
        
    }

    return 0;
}
