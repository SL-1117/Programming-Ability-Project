static unsigned char FrameBuf[LCD_Xmax * (LCD_Ymax / 8)];   // shifted frame to be displayed

/* ---- BEGIN SCROLL ---- */
// Shift the source image by (ox, oy) pixels with wrap-around, result goes to FrameBuf
// Format: byte index = x + page*128, bit0 = top pixel of the page
static void build_frame(int ox, int oy)
{
    int x, y, sx, sy;
    for (x = 0; x < LCD_Xmax * (LCD_Ymax / 8); x++) FrameBuf[x] = 0;

    for (y = 0; y < LCD_Ymax; y++) {
        sy = (y - oy + LCD_Ymax) % LCD_Ymax;                  // source y
        for (x = 0; x < LCD_Xmax; x++) {
            sx = (x - ox + LCD_Xmax) % LCD_Xmax;              // source x
            if (BMP_Logo_Nuvoton[sx + (sy >> 3) * LCD_Xmax] & (1 << (sy & 7)))
                FrameBuf[x + (y >> 3) * LCD_Xmax] |= (1 << (y & 7));
        }
    }
}
/* ---- END SCROLL ---- */

// Wait until the key is released (with simple debounce)
static void wait_release(void)
{
    CLK_SysTickDelay(20000);
    while (ScanKey() != 0);
    CLK_SysTickDelay(20000);
}

int main(void)
{
    int8_t  dirX = 0, dirY = 0;     // current direction, both 0 = stopped
    int16_t ox = 0, oy = 0;         // current offset
    uint8_t key;

    SYS_Init();
    init_LCD();
    clear_LCD();
    OpenKeyPad();

    OpenSevenSegment();
    CloseSevenSegment();
    ShowSevenSegment(3, 5);         // test: show 5 on the rightmost digit (static output)

    draw_LCD(BMP_Logo_Nuvoton);     // initial state: show LOGO

    while (1) {
        key = ScanKey();
        if (key != 0) {
            wait_release();         // act only after key release
            switch (key) {
                case 1: dirX = -1; dirY = -1; break;
                case 2: dirX =  0; dirY = -1; break;
                case 3: dirX = +1; dirY = -1; break;
                case 4: dirX = -1; dirY =  0; break;
                case 5: dirX =  0; dirY =  0; break;   // S : stop
                case 6: dirX = +1; dirY =  0; break;
                case 7: dirX = -1; dirY = +1; break;
                case 8: dirX =  0; dirY = +1; break;
                case 9: dirX = +1; dirY = +1; break;
                default: break;
            }
        }

        if (dirX != 0 || dirY != 0) {
            ox = (ox + dirX * STEP + LCD_Xmax) % LCD_Xmax;
            oy = (oy + dirY * STEP + LCD_Ymax) % LCD_Ymax;
            build_frame(ox, oy);
            draw_LCD(FrameBuf);     // redraw only when moving; frame is static when stopped
        }
        CLK_SysTickDelay(MOVE_DELAY);
    }
}
