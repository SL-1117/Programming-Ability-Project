#include <stdio.h>
#include <stdlib.h>

#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "Seven_Segment.h"
#include "Scankey.h"

#define C 8
#define R 9

// RGB LED: Active LOW
#define RGB_BLUE  PA12
#define RGB_GREEN PA13
#define RGB_RED   PA14

void Display_7seg(uint16_t value)
{
    uint8_t tens;
    uint8_t ones;

    tens = value / 10;
    ones = value % 10;

    // Display tens digit if needed
    if (tens != 0)
    {
        CloseSevenSegment();
        ShowSevenSegment(1, tens);
        CLK_SysTickDelay(5000);
    }

    // Display ones digit
    CloseSevenSegment();
    ShowSevenSegment(0, ones);
    CLK_SysTickDelay(5000);
}

void Clear_7seg(void)
{
    CloseSevenSegment();
    CLK_SysTickDelay(5000);
}

int IsPrime(int n)
{
    int i;

    if (n < 2)
    {
        return 0;
    }

    for (i = 2; i * i <= n; i++)
    {
        if (n % i == 0)
        {
            return 0;
        }
    }

    return 1;
}

void RGB_Clear(void)
{
    RGB_RED = 1;
    RGB_GREEN = 1;
    RGB_BLUE = 1;
}

void Update_RGB(int num)
{
    RGB_Clear();

    // Prime -> RED
    if (IsPrime(num))
    {
        RGB_RED = 0;
    }

    // Even -> BLUE
    if (num % 2 == 0)
    {
        RGB_BLUE = 0;
    }

    // Multiple of 3 -> GREEN
    if (num % 3 == 0)
    {
        RGB_GREEN = 0;
    }
}

void Buzz(int number, int displayNum)
{
    int i;

    for (i = 0; i < number; i++)
    {
        // Buzzer ON
        PB11 = 0;
        Display_7seg(displayNum);

        // Buzzer OFF
        PB11 = 1;
        Display_7seg(displayNum);
    }
}

int main(void)
{
    int key;
    int lastKey = 0;
    int num = 0;

    SYS_Init();

    OpenSevenSegment();
    OpenKeyPad();

    // Buzzer
    GPIO_SetMode(PB, BIT11, GPIO_MODE_OUTPUT);
    PB11 = 1;

    // RGB LED
    GPIO_SetMode(PA, BIT12, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT13, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT14, GPIO_MODE_OUTPUT);

    RGB_Clear();

    // Random seed
    srand(SysTick->VAL);

    Clear_7seg();
    RGB_Clear();
    PB11 = 1;

    while (1)
    {
        key = ScanKey();

        // Detect R release
        if (lastKey == R && key == 0)
        {
            num = rand() % 99 + 1;

            Update_RGB(num);

            Display_7seg(num);

            // Prime -> beep once
            if (IsPrime(num))
            {
                Buzz(1, num);
            }
        }

        // Detect C release
        if (lastKey == C && key == 0)
        {
            num = 0;

            Clear_7seg();
            RGB_Clear();
            PB11 = 1;
        }

        // Keep displaying the current number
        if (num != 0)
        {
            Display_7seg(num);
        }

        lastKey = key;
    }
}

#define BUZZER PB11

#define BLUE PA12

void Buzz(void);

int IsPrime(int num);

void ClearALL(void);

int main(void)
{
    int key = 0;
    int last_key = 0;
    int num = 0;
    
    SYS_Init();

    OpenSevenSegment();
    OpenKeyPad();

    GPIO_SetMode(PA, BIT12, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT13, GPIO_MODE_OUTPUT);
    GPIO_SetMode(PA, BIT14, GPIO_MODE_OUTPUT);

    GPIO_SetMode(PB, BIT11, GPIO_MODE_OUTPUT);

    BUZZER = 1;

    srand(SysTick -> VAL);

    while (1)
    {
        key = Scankey();

        if (last_key == C && key == 0)
        {
            num = 0;
            ClearALL();
        }

        if (last_key == R && key == 0)
        {
            num = rand() % 99 + 1;

            if (IsPrime(num))
            {
                RED = 0;

                if (num % 2 == 0) BLUE = 0;
                if (num % 3 == 0) GREEN = 0;
                
                Buzz();
            }

            else 
            {
                if (num % 2 == 0) BLUE = 0;
                if (num % 3 == 0) GREEN = 0;
            }
        }

        Show_7segment(num);

        last_key = key;
    }

}

void Show_7segment(int num)
{
    static int digit = 0;

    int d0 = num % 10; 
    int d1 = (num / 10) % 10;

    CloseSevenSegment();

    if (digit == 0 && num != 0) ShowSevenSegment(0, d0);
    if (digit == 1 && num >= 10) ShowSevenSegment(1, d1);

    digit++;

    if (digit >= 2) digit = 0;
}

void Buzz(void)
{
    BUZZER = 0;
    CLK_SysTickDelay(2000);

    BUZZER = 1;
    CLK_SysTickDelay(2000);
}

int IsPrime(int num)
{
    if (num < 2) return 0;

    for (int i = 2; i * i <= num; i++)
    {
        if (num % i == 0)
        {
            return 0;
        }
    }

    return 1;
}

void ClearALL(void)
{
    BUZZER = 1;
    RED = 1;
    BLUE = 1;
    GREEN = 1;
    CloseSevenSegment();
}
