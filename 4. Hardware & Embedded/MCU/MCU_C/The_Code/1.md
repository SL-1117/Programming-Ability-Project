#include <stdio.h>

#include "MCU.h"
#include "SYS.h"
#include "GPIO.h"
#include "NCU100Series.h"
#include "Scankey.h"
#include "Seven_Segment.h"

int main(void)
{
    SYS_Init();

    GPIO_SetMode(PC, BIT12, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT13, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT14, GPIO_PMD_OUTPUT);
    GPIO_SetMode(PC, BIT15, GPIO_PMD_OUTPUT);

    PC12 = 1;
    PC13 = 1;
    PC14 = 1;
    PC15 = 1;

    while(1)
    {
        PC15 = 0;
        CLK_SysTickDelay(2000000);
        PC15 = 1;
        
        PC14 = 0;
        CLK_SysTickDelay(2000000);
        PC14 = 1;

        PC13 = 0;
        CLK_SysTickDelay(2000000);
        PC13 = 1;

        PC12 = 0;
        CLK_SysTickDelay(2000000);
        PC12 = 1;
    }

    return 0;
}
